SELECT
  a.AlertId,
  a.SensorId,
  a.Status,
  a.OpenedAt,
  a.AdjustedAtOpen,
  a.MinAtOpen,
  a.MaxAtOpen,
  a.SnoozedUntil,
  a.Message,
  s.Lab,
  s.Department
FROM AlertLog a
JOIN Sensor s
  ON CONCAT(s.HardwareName, ':', s.LOC) = a.SensorId
WHERE a.Status IN ('OPEN','ACK')
  AND (a.SnoozedUntil IS NULL OR a.SnoozedUntil < NOW())

  -- Lab filter (only added when a lab is picked)
  {{ SelectLab.selectedOptionValue
      ? "AND LOWER(TRIM(s.Lab)) = LOWER(TRIM('" + SelectLab.selectedOptionValue + "'))"
      : "" }}

  -- Department filter (only added when a dept is picked)
  {{ SelectDept.selectedOptionValue
      ? "AND LOWER(TRIM(s.Department)) = LOWER(TRIM('" + SelectDept.selectedOptionValue + "'))"
      : "" }}

ORDER BY a.OpenedAt DESC;
