SELECT 
  sd.ID,
  sd.device_id,
  sd.Lab,
  sd.Dept,
  sd.locality,
  sd.Value,
  sd.Timestamp,
  sd.wifi_signal,
  sd.config_version,
  sd.pam_service_version,
	sd.Serial
FROM SensorData sd
JOIN (
  SELECT 
    ID,
    MAX(`Timestamp`) AS ts
  FROM SensorData
  WHERE '{{ LabSelect.selectedOptionValue || "" }}' <> ''
    AND Lab = '{{ LabSelect.selectedOptionValue }}'
    AND ('{{ DeptSelect.selectedOptionValue || "ALL" }}' = 'ALL'
         OR Dept = '{{ DeptSelect.selectedOptionValue }}')
    AND `Timestamp` >= NOW() - INTERVAL 24 HOUR
  GROUP BY ID
) x ON x.ID = sd.ID AND x.ts = sd.`Timestamp`
ORDER BY sd.ID;
