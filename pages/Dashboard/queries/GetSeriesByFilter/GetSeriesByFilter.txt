SELECT
  sd.ID          AS SensorKey,
  sd.Timestamp,
  sd.Value,
	sd.Lab				AS Locationab
FROM SensorData sd
JOIN Sensor s
  ON s.HardwareName = sd.ID
WHERE s.Active = 1
  AND '{{ JS_Dash.lab() }}' <> ''                 -- guard: require Lab
  AND s.Lab = '{{ JS_Dash.lab() }}'
  AND (
        '{{ JS_Dash.dept() }}' = ''               -- if no Dept selected -> include all
        OR s.Department = '{{ JS_Dash.dept() }}'
      )
  AND sd.Timestamp >= NOW() - INTERVAL 24 HOUR
ORDER BY sd.Timestamp ASC;
