WITH fs AS (
  SELECT HardwareName, Lab, Department
  FROM Sensor
  WHERE Active = 1
    AND '{{ JS_Dash.lab() }}' <> ''               -- guard: require Lab
    AND Lab = '{{ JS_Dash.lab() }}'
    AND (
          '{{ JS_Dash.dept() }}' = ''             -- if no Dept selected -> include all
          OR Department = '{{ JS_Dash.dept() }}'
        )
),
mx AS (
  SELECT sd.ID, MAX(sd.Timestamp) AS MaxTs
  FROM SensorData sd
  JOIN fs ON fs.HardwareName = sd.ID AND sd.Lab = fs.Lab
  GROUP BY sd.ID
)
SELECT
  fs.HardwareName   AS SensorKey,
  sd.Value,
  sd.Timestamp
FROM mx
JOIN SensorData sd ON sd.ID = mx.ID AND sd.Timestamp = mx.MaxTs
JOIN fs ON fs.HardwareName = sd.ID
ORDER BY fs.HardwareName;
